class PythonListImpl:
    class Node:
        def __init__(self, data):
            self.data = data
            self.next = None

    def __init__(self):
        self.head = None

    def create(self): pass
    def destroy(self): self.head = None
    def insert_beginning(self, val):
        new_node = self.Node(val)
        new_node.next = self.head
        self.head = new_node

    def insert_end(self, val):
        new_node = self.Node(val)
        if not self.head:
            self.head = new_node
            return
        curr = self.head
        while curr.next:
            curr = curr.next
        curr.next = new_node

    def delete_beginning(self):
        if self.head: self.head = self.head.next

    def delete_end(self):
        if not self.head: return
        if not self.head.next:
            self.head = None
            return
        curr = self.head
        while curr.next.next:
            curr = curr.next
        curr.next = None

    def get_elements(self):
        res = []
        curr = self.head
        while curr:
            res.append(curr.data)
            curr = curr.next
        return res


class CtypesListImpl:
    #
    # 
    #
    def __init__(self):
        base_dir = os.path.dirname(os.path.abspath(__file__))
        # Ищем либо новую clist.dll, либо старую DoublyLinkedList.dll
        clist_path = os.path.join(base_dir, "clist.dll")
        doubly_path = os.path.join(base_dir, "DoublyLinkedList.dll")
        
        if os.path.exists(clist_path):
            lib_name = clist_path
        elif os.path.exists(doubly_path):
            lib_name = doubly_path
        else:
            raise FileNotFoundError("Библиотека clist.dll или DoublyLinkedList.dll не найдена! Положите её рядом с .py скриптом.")
        
        self.lib = ctypes.CDLL(lib_name)
        
        # Описание структуры узла С в памяти
        class OneLinkedList(ctypes.Structure):
            pass
        OneLinkedList._fields_ = [("data", ctypes.c_double), ("next", ctypes.POINTER(OneLinkedList))]
        
        self.NodePtr = ctypes.POINTER(OneLinkedList)
        self.NodePtrPtr = ctypes.POINTER(self.NodePtr)
        
        # Настройка сигнатур базовых функций
        self.lib.insert_beginning.argtypes = [self.NodePtrPtr, ctypes.c_double]
        self.lib.insert_end.argtypes = [self.NodePtrPtr, ctypes.c_double]
        self.lib.delete_beginning.argtypes = [self.NodePtrPtr]
        self.lib.delete_end.argtypes = [self.NodePtrPtr]
        self.lib.delete_list.argtypes = [self.NodePtrPtr]
        self.lib.init_node.argtypes = [self.NodePtrPtr]

        # Указатель на голову списка (изначально NULL)
        self.head = self.NodePtr()

    def create(self): 
        self.lib.init_node(ctypes.byref(self.head))
        
    def destroy(self): 
        self.lib.delete_list(ctypes.byref(self.head))
        
    def insert_beginning(self, val): 
        self.lib.insert_beginning(ctypes.byref(self.head), ctypes.c_double(val))
        
    def insert_end(self, val): 
        self.lib.insert_end(ctypes.byref(self.head), ctypes.c_double(val))
        
    def delete_beginning(self): 
        self.lib.delete_beginning(ctypes.byref(self.head))
        
    def delete_end(self): 
        self.lib.delete_end(ctypes.byref(self.head))

    def get_elements(self):
        # Читаем список напрямую из памяти C, переходя по указателям!
        # Это избавляет от необходимости писать функции get_length/to_array в C
        res = []
        curr = self.head
        while curr:
            node = curr.contents
            res.append(node.data)
            curr = node.next # Переход к следующему узлу
        return res


class PyBindListImpl:
    """Реализация через C++ (pybind11)"""
    def __init__(self):
        if not PYBIND_AVAILABLE:
            raise RuntimeError("Модуль pybind_list не скомпилирован!")
        self.obj = pybind_list.SinglyLinkedList()

    def create(self): self.obj.init_node()
    def destroy(self): self.obj.delete_list()
    def insert_beginning(self, val): self.obj.insert_beginning(val)
    def insert_end(self, val): self.obj.insert_end(val)
    def delete_beginning(self): self.obj.delete_beginning()
    def delete_end(self): self.obj.delete_end()
    def get_elements(self): return self.obj.get_elements()


# ==================== интерфейсикс ====================

class App(ctk.CTk):
    def __init__(self):
        super().__init__()
        self.title("Лабораторная: Односвязный список")
        self.geometry("650x500")
        ctk.set_appearance_mode("dark")

        self.list_impl = None
        self.current_mode = "Python"

        self.build_ui()
        self.update_display()

    def build_ui(self):
        
        self.lbl_list_title = ctk.CTkLabel(self, text="Содержимое списка:", font=("Arial", 16, "bold"))
        self.lbl_list_title.pack(pady=(20, 5))

        self.textbox_list_content = ctk.CTkTextbox(
            self, 
            width=600, 
            height=150, 
            font=("Consolas", 18), 
            wrap="word",
            activate_scrollbars=True
        )
        self.textbox_list_content.pack(pady=5, padx=20, fill="x")
        self.textbox_list_content.configure(state="disabled")

        
        frm_input = ctk.CTkFrame(self, fg_color="transparent")
        frm_input.pack(pady=10)

        ctk.CTkLabel(frm_input, text="Значение:", font=("Arial", 14)).grid(row=0, column=0, padx=5)
        self.entry_val = ctk.CTkEntry(frm_input, width=150, font=("Arial", 14))
        self.entry_val.grid(row=0, column=1, padx=5)

        
        frm_btn = ctk.CTkFrame(self, fg_color="transparent")
        frm_btn.pack(pady=5)

        self.btn_create = ctk.CTkButton(frm_btn, text="Создать список", command=self.create_list, width=150)
        self.btn_create.grid(row=0, column=0, padx=5, pady=5)
        
        self.btn_destroy = ctk.CTkButton(frm_btn, text="Удалить список", command=self.destroy_list, width=150, fg_color="#D32F2F", hover_color="#B71C1C")
        self.btn_destroy.grid(row=0, column=1, padx=5, pady=5)

        self.btn_add_beg = ctk.CTkButton(frm_btn, text="Добавить в начало", command=lambda: self.add_item("beg"), width=150)
        self.btn_add_beg.grid(row=1, column=0, padx=5, pady=5)
        
        self.btn_add_end = ctk.CTkButton(frm_btn, text="Добавить в конец", command=lambda: self.add_item("end"), width=150)
        self.btn_add_end.grid(row=1, column=1, padx=5, pady=5)

        self.btn_del_beg = ctk.CTkButton(frm_btn, text="Удалить в начале", command=lambda: self.del_item("beg"), width=150)
        self.btn_del_beg.grid(row=2, column=0, padx=5, pady=5)
        
        self.btn_del_end = ctk.CTkButton(frm_btn, text="Удалить в конце", command=lambda: self.del_item("end"), width=150)
        self.btn_del_end.grid(row=2, column=1, padx=5, pady=5)

        
        frm_radio = ctk.CTkFrame(self)
        frm_radio.pack(pady=10, fill="x", padx=20)

        ctk.CTkLabel(frm_radio, text="Способ реализации:", font=("Arial", 14, "bold")).pack(pady=5)

        self.radio_var = ctk.StringVar(value="Python")
        modes = ["Python", "C++(PyBind)", "C(ctypes)"]
        for mode in modes:
            ctk.CTkRadioButton(frm_radio, text=mode, variable=self.radio_var, value=mode, command=self.on_mode_change).pack(side="left", padx=15, pady=5)

    def update_display(self):
        self.textbox_list_content.configure(state="normal")
        self.textbox_list_content.delete("1.0", "end")
        
        if self.list_impl is None:
            self.textbox_list_content.insert("end", "Сначала список создай, а потом фигней страдай!")
        else:
            elements = self.list_impl.get_elements()
            if not elements:
                self.textbox_list_content.insert("end", "Список пуст, более не скажу.")
            else:
                # ИСПРАВЛЕНО: Округление убрано, числа выводятся как есть
                str_elems = [str(int(e)) if e.is_integer() else str(e) for e in elements]
                self.textbox_list_content.insert("end", " -> ".join(str_elems))
                
        self.textbox_list_content.configure(state="disabled")

    def get_val(self):
        try:
            return float(self.entry_val.get())
        except ValueError:
            self.show_error("ты ввел фигню, а не ЧИСЛО!")
            return None

    def create_list(self):
        if self.list_impl is not None:
            self.show_error("Ты же уже создал список. Сразу два мы не поддерживаем, удали старый.")
            return
        
        try:
            self.list_impl = self.get_impl_class()()
            self.list_impl.create()
            self.update_display()
        except Exception as e:
            self.show_error(str(e))

    def destroy_list(self):
        if self.list_impl is None:
            return
        self.list_impl.destroy()
        self.list_impl = None
        self.update_display()

    def add_item(self, pos):
        if not self.check_created(): return
        val = self.get_val()
        if val is None: return
        
        if pos == "beg": self.list_impl.insert_beginning(val)
        else: self.list_impl.insert_end(val)
        self.update_display()

    def del_item(self, pos):
        if not self.check_created(): return
        if pos == "beg": self.list_impl.delete_beginning()
        else: self.list_impl.delete_end()
        self.update_display()

    def check_created(self):
        if self.list_impl is None:
            self.show_error("Сначала список создай, а потом фигней страдай!")
            return False
        return True

    def get_impl_class(self):
        mode = self.radio_var.get()
        if mode == "Python": return PythonListImpl
        if mode == "C++(PyBind)": return PyBindListImpl
        if mode == "C(ctypes)": return CtypesListImpl

    def on_mode_change(self):
        new_mode = self.radio_var.get()
        if self.list_impl is not None and new_mode != self.current_mode:
            msg = "В списке уже есть элементы, что же можно с ними сделать?\n\nДа - Перенести весь список в новую реализацию\nНет - Удалить список\nОтмена - остаться на текущей реализации"
            res = messagebox.askyesnocancel("Переключение реализации", msg)
            
            if res is True: # Да (Перенести)
                elements = self.list_impl.get_elements()
                self.list_impl.destroy()
                try:
                    self.list_impl = self.get_impl_class()()
                    self.list_impl.create()
                    for item in elements:
                        self.list_impl.insert_end(item)
                except Exception as e:
                    self.show_error(str(e))
                    self.list_impl = None
                    self.radio_var.set(self.current_mode)
                    self.update_display()
                    return
            elif res is False: # Нет (Удалить)
                self.list_impl.destroy()
                self.list_impl = None
            else: # Отмена
                self.radio_var.set(self.current_mode)
                return
        
        self.current_mode = self.radio_var.get()
        self.update_display()

    def show_error(self, msg):
        messagebox.showerror("Ошибка", msg)

if __name__ == "__main__":
    app = App()
    app.mainloop()
